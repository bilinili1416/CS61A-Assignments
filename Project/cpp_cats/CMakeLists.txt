cmake_minimum_required(VERSION 3.16)
project(cats_typing_test VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(cats_lib cats.cpp)
target_include_directories(cats_lib PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")

add_executable(cats_cpp main.cpp)
target_link_libraries(cats_cpp PRIVATE cats_lib)
target_compile_definitions(cats_cpp PRIVATE CATS_DATA_DIR="${CMAKE_CURRENT_SOURCE_DIR}/data")

include(CTest)
if(BUILD_TESTING)
    add_executable(cats_test cats_test.cpp)
    target_link_libraries(cats_test PRIVATE cats_lib)
    add_test(NAME cats_test COMMAND cats_test)
endif()

if(MSVC)
    target_compile_options(cats_lib PRIVATE /W4 /permissive-)
    target_compile_options(cats_cpp PRIVATE /W4 /permissive-)
    if(BUILD_TESTING)
        target_compile_options(cats_test PRIVATE /W4 /permissive-)
    endif()
else()
    target_compile_options(cats_lib PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(cats_cpp PRIVATE -Wall -Wextra -Wpedantic)
    if(BUILD_TESTING)
        target_compile_options(cats_test PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endif()
